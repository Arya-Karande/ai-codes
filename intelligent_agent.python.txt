
# Intelligent Number Guessing Agent

low = 1
high = 100
attempts = 0

print("Think of a number between 1 and 100.")
print("I will try to guess your number!")
print("Enter: h = My guess is higher")
print("       l = My guess is lower")
print("       c = My guess is correct")

while low <= high:
    guess = (low + high) // 2
    attempts += 1

    print("\nIs your number", guess, "?")
    feedback = input("Enter h, l, or c: ").lower().strip()

    if feedback == 'c':
        print("Agent: I guessed your number!")
        print("Number of attempts:", attempts)
        break

    elif feedback == 'h':
        # Guess is too high
        high = guess - 1

    elif feedback == 'l':
        # Guess is too low
        low = guess + 1

    else:
        print("Invalid feedback! Enter h, l, or c.")
        attempts -= 1

else:
    print("Feedback is inconsistent or the number is outside the range.")
