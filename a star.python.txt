
import heapq

# Goal state
goal = (1, 2, 3,
        4, 5, 6,
        7, 8, 0)

# Manhattan Distance Heuristic
def heuristic(state):
    distance = 0

    for i in range(9):
        if state[i] != 0:
            goal_pos = state[i] - 1

            x1, y1 = i // 3, i % 3
            x2, y2 = goal_pos // 3, goal_pos % 3

            distance += abs(x1 - x2) + abs(y1 - y2)

    return distance

# Generate neighboring states
def get_neighbors(state):
    neighbors = []
    zero = state.index(0)
    row, col = zero // 3, zero % 3

    moves = [(-1, 0), (1, 0),
             (0, -1), (0, 1)]

    for dr, dc in moves:
        r, c = row + dr, col + dc

        if 0 <= r < 3 and 0 <= c < 3:
            new_pos = r * 3 + c
            new_state = list(state)

            new_state[zero], new_state[new_pos] = (
                new_state[new_pos], new_state[zero]
            )

            neighbors.append(tuple(new_state))

    return neighbors

# A* Search
def astar(start):
    pq = [(heuristic(start), 0, start)]
    parent = {start: None}
    cost = {start: 0}

    while pq:
        f, g, current = heapq.heappop(pq)

        if g != cost[current]:
            continue

        if current == goal:
            path = []

            while current is not None:
                path.append(current)
                current = parent[current]

            return path[::-1]

        for neighbor in get_neighbors(current):
            new_g = g + 1

            if neighbor not in cost or new_g < cost[neighbor]:
                cost[neighbor] = new_g
                parent[neighbor] = current

                new_f = new_g + heuristic(neighbor)
                heapq.heappush(
                    pq, (new_f, new_g, neighbor)
                )

    return None

# Take user input
print("Enter the initial state (use 0 for blank):")
start = []

for i in range(3):
    row = list(map(int, input().split()))
    start.extend(row)

start = tuple(start)

# Validate input
if len(start) != 9 or set(start) != set(range(9)):
    print("Invalid input! Enter numbers 0 to 8 exactly once.")
else:
    # Check solvability
    arr = [x for x in start if x != 0]
    inversions = sum(
        1 for i in range(len(arr))
        for j in range(i + 1, len(arr))
        if arr[i] > arr[j]
    )

    if inversions % 2 != 0:
        print("This puzzle is not solvable.")
    else:
        path = astar(start)

        if path:
            print("\nSolution found!")
            print("Minimum moves:", len(path) - 1)

            for step, state in enumerate(path):
                print("\nStep", step)
                for i in range(0, 9, 3):
                    print(state[i:i+3])
        else:
            print("No solution found.")
