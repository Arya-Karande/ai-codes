
from collections import deque

# Goal state
goal = (1, 2, 3,
        4, 5, 6,
        7, 8, 0)

# Generate all possible next states
def get_neighbors(state):
    neighbors = []
    zero = state.index(0)
    row, col = zero // 3, zero % 3

    moves = [(-1, 0), (1, 0),
             (0, -1), (0, 1)]

    for dr, dc in moves:
        r, c = row + dr, col + dc

        if 0 <= r < 3 and 0 <= c < 3:
            new_pos = r * 3 + c
            new_state = list(state)

            new_state[zero], new_state[new_pos] = (
                new_state[new_pos], new_state[zero]
            )

            neighbors.append(tuple(new_state))

    return neighbors


# BFS algorithm
def bfs(start):
    queue = deque([start])
    visited = {start}
    parent = {start: None}

    while queue:
        current = queue.popleft()

        if current == goal:
            path = []

            while current is not None:
                path.append(current)
                current = parent[current]

            return path[::-1]

        for neighbor in get_neighbors(current):
            if neighbor not in visited:
                visited.add(neighbor)
                parent[neighbor] = current
                queue.append(neighbor)

    return None


# Check whether the puzzle is solvable
def is_solvable(state):
    arr = [x for x in state if x != 0]
    inversions = sum(
        1 for i in range(len(arr))
        for j in range(i + 1, len(arr))
        if arr[i] > arr[j]
    )
    return inversions % 2 == 0


# Take input from user
print("Enter the initial state row by row (use 0 for blank):")
start = []

for i in range(3):
    start.extend(map(int, input().split()))

start = tuple(start)

if len(start) != 9 or set(start) != set(range(9)):
    print("Invalid input! Enter numbers 0 to 8 exactly once.")
elif not is_solvable(start):
    print("This puzzle is not solvable.")
else:
    path = bfs(start)

    if path:
        print("\nSolution found!")
        print("Minimum moves:", len(path) - 1)

        for step, state in enumerate(path):
            print("\nStep", step)
            for i in range(0, 9, 3):
                print(*state[i:i+3])
    else:
        print("No solution found.")
