
# Tic-Tac-Toe using Minimax Algorithm

import math

board = [" "] * 9

def print_board():
    for i in range(0, 9, 3):
        print("|".join(board[i:i+3]))
        if i < 6:
            print("---------")

def check_winner(player):
    wins = [
        (0, 1, 2), (3, 4, 5), (6, 7, 8),
        (0, 3, 6), (1, 4, 7), (2, 5, 8),
        (0, 4, 8), (2, 4, 6)
    ]

    return any(
        board[a] == board[b] == board[c] == player
        for a, b, c in wins
    )

def minimax(is_maximizing):
    if check_winner("O"):
        return 1

    if check_winner("X"):
        return -1

    if " " not in board:
        return 0

    if is_maximizing:
        best = -math.inf

        for i in range(9):
            if board[i] == " ":
                board[i] = "O"
                score = minimax(False)
                board[i] = " "
                best = max(best, score)

        return best

    else:
        best = math.inf

        for i in range(9):
            if board[i] == " ":
                board[i] = "X"
                score = minimax(True)
                board[i] = " "
                best = min(best, score)

        return best

def best_move():
    best_score = -math.inf
    move = -1

    for i in range(9):
        if board[i] == " ":
            board[i] = "O"
            score = minimax(False)
            board[i] = " "

            if score > best_score:
                best_score = score
                move = i

    return move

# Game starts
print("Tic-Tac-Toe")
print("You are X, Computer is O")
print("Enter positions from 1 to 9:")

print("1|2|3")
print("-----")
print("4|5|6")
print("-----")
print("7|8|9")

while True:
    try:
        pos = int(input("\nEnter your position (1-9): ")) - 1
    except ValueError:
        print("Enter a valid number.")
        continue

    if pos < 0 or pos > 8 or board[pos] != " ":
        print("Invalid move! Try again.")
        continue

    board[pos] = "X"
    print_board()

    if check_winner("X"):
        print("You win!")
        break

    if " " not in board:
        print("It's a draw!")
        break

    print("\nComputer is thinking...")
    move = best_move()
    board[move] = "O"
    print_board()

    if check_winner("O"):
        print("Computer wins!")
        break

    if " " not in board:
        print("It's a draw!")
        break
