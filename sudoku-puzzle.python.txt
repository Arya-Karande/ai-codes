
# Sudoku Solver using Backtracking and Constraint Propagation

def get_candidates(board, row, col):
    # Find possible numbers for an empty cell
    used = set(board[row])

    for i in range(9):
        used.add(board[i][col])

    start_row = (row // 3) * 3
    start_col = (col // 3) * 3

    for i in range(start_row, start_row + 3):
        for j in range(start_col, start_col + 3):
            used.add(board[i][j])

    return set(range(1, 10)) - used


def solve(board):
    # Constraint propagation: find cell with fewest candidates
    best_cell = None
    best_candidates = None

    for row in range(9):
        for col in range(9):
            if board[row][col] == 0:
                candidates = get_candidates(board, row, col)

                if not candidates:
                    return False

                if (best_candidates is None or
                        len(candidates) < len(best_candidates)):
                    best_cell = (row, col)
                    best_candidates = candidates

    # No empty cells means Sudoku is solved
    if best_cell is None:
        return True

    row, col = best_cell

    # Backtracking: try each possible number
    for num in sorted(best_candidates):
        board[row][col] = num

        if solve(board):
            return True

        # Undo the move if it fails
        board[row][col] = 0

    return False


# Take user input
print("Enter Sudoku row by row (use 0 for empty cells):")
board = []

for i in range(9):
    row = list(map(int, input().split()))

    if len(row) != 9 or any(num < 0 or num > 9 for num in row):
        print("Invalid input! Each row must contain 9 numbers from 0 to 9.")
        raise SystemExit

    board.append(row)


# Check for duplicate numbers in initial clues
def valid_board(board):
    for r in range(9):
        for c in range(9):
            num = board[r][c]

            if num != 0:
                board[r][c] = 0
                candidates = get_candidates(board, r, c)
                board[r][c] = num

                if num not in candidates:
                    return False

    return True


if not valid_board(board):
    print("Invalid Sudoku puzzle!")
elif solve(board):
    print("\nSolved Sudoku:")
    for row in board:
        print(*row)
else:
    print("No solution exists.")
