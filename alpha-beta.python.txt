
# Alpha-Beta Pruning Algorithm

def alpha_beta(depth, node, maximizing, values,
               alpha, beta, max_depth):

    # Leaf node
    if depth == max_depth:
        return values[node]

    if maximizing:
        best = float('-inf')

        for child in range(2):
            value = alpha_beta(
                depth + 1,
                node * 2 + child,
                False,
                values,
                alpha,
                beta,
                max_depth
            )

            best = max(best, value)
            alpha = max(alpha, best)

            # Alpha-Beta Pruning
            if beta <= alpha:
                break

        return best

    else:
        best = float('inf')

        for child in range(2):
            value = alpha_beta(
                depth + 1,
                node * 2 + child,
                True,
                values,
                alpha,
                beta,
                max_depth
            )

            best = min(best, value)
            beta = min(beta, best)

            # Alpha-Beta Pruning
            if beta <= alpha:
                break

        return best


# User input
n = int(input("Enter number of leaf nodes (power of 2): "))
values = list(map(int, input("Enter leaf node values: ").split()))

# Validate input
if n != len(values) or n < 2 or (n & (n - 1)) != 0:
    print("Invalid input! Enter a power of 2 and matching values.")
else:
    max_depth = n.bit_length() - 1

    result = alpha_beta(
        0, 0, True, values,
        float('-inf'), float('inf'),
        max_depth
    )

    print("Optimal value:", result)
